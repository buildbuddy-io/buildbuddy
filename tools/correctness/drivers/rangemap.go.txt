package rangemap

import (
	"fmt"
	"os"
	"testing"
	"unicode/utf8"
)

func guarded(t *testing.T) {
	if p := recover(); p != nil {
		t.Fatalf("contract panic: %v", p)
	}
}

// The reference model uses integer interval arithmetic, not binary search.
func TestRangeModel(t *testing.T) {
	for _, count := range []int{0, 1, 2, 3} {
		for start := 0; start < 10; start++ {
			for end := start + 1; end <= 10; end++ {
				t.Run(fmt.Sprintf("count=%d/query=%d-%d", count, start, end), func(t *testing.T) {
					defer guarded(t)
					m := @NEW@
					for i := 0; i < count; i++ {
						if _, err := m.Add([]byte{byte(1 + 3*i)}, []byte{byte(3 + 3*i)}, i+1); err != nil {
							t.Fatal(err)
						}
					}
					for key := 0; key <= 10; key++ {
						want := 0
						for i := 0; i < count; i++ {
							if key >= 1+3*i && key < 3+3*i {
								want = i + 1
							}
						}
						@LOOKUP@
						if got != want {
							t.Fatalf("lookup key=%d got=%d want=%d", key, got, want)
						}
					}
					if os.Getenv("HARNESS_PROFILE") == "baseline" {
						return
					}
					want := []int{}
					for i := 0; i < count; i++ {
						if start < 3+3*i && 1+3*i < end {
							want = append(want, i+1)
						}
					}
					ranges := m.GetOverlapping([]byte{byte(start)}, []byte{byte(end)})
					got := []int{}
					for _, r := range ranges {
						got = append(got, @VALUE@)
					}
					if fmt.Sprint(got) != fmt.Sprint(want) {
						t.Fatalf("overlap got=%v want=%v", got, want)
					}
				})
			}
		}
	}
}

// Human-readable diagnostics must remain valid text for arbitrary binary keys.
func TestRangeDiagnosticText(t *testing.T) {
	if os.Getenv("HARNESS_PROFILE") == "baseline" {
		t.Skip("binary input and diagnostic oracle absent")
	}
	for b := 0; b < 256; b++ {
		t.Run(fmt.Sprintf("byte=%d", b), func(t *testing.T) {
			defer guarded(t)
			m := @NEW@
			r, err := m.Add([]byte{byte(b)}, []byte{byte(b), 0}, 1)
			if err != nil {
				t.Fatal(err)
			}
			if !utf8.ValidString(r.String()) {
				t.Fatalf("diagnostic is invalid UTF-8: %q", r.String())
			}
		})
	}
}
